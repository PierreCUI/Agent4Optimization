The design domain $D$ is the rectangle $[0,4]\times[0,1]$ (the square() mesh with implicit labels: 1 = bottom, 2 = right, 3 = top, 4 = left).  The vertical load $f$ acts on the top edge $\Gamma_N$ (label 3).  The left and right edges $\Gamma_{2}\cup\Gamma_{4}$ (labels 2, 4) are the Dirichlet supports and are FROZEN (fixedLabels=[2,4]); the bottom and top edges (labels 1, 3) are NORMAL-LOCKED (boundaryLabels=[1,3]); no boundary is free.
Let $\phi(x)$ be the level-set function and
 \begin{align*}\raisebox{\depth}{$\chi$}(x) &= \begin{cases} 1 & \mbox{if}\quad \phi(x) < 0 \\ 1/1000 & \mbox{if}\quad \phi(x)  \ge 0 \end{cases}.\end{align*}
The Lagrangian of the linear Elasticity bridge problem is
\begin{dmath*}
\mathcal L(\left[ {\mathbf{ u }}\right],\left[ {\mathbf{ v }}\right]) = J({\mathbf{ u }}) -  \int_{\Gamma_N} f \left( \mathbf e_y \cdot {\mathbf{ v }} \right) \,ds_x +  \int_{D} \raisebox{\depth}{$\chi$} \left(\lambda \operatorname{div} {\mathbf{ v }}  \,\operatorname{tr}\left(\pmb \varepsilon({\mathbf{ u }})\right)  + 2 \mu \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \nabla {\mathbf{ v }}  \right)\right) \,dx , 
\end{dmath*}
with 
\begin{align*}
J({\mathbf{ u }}) &=  \int_{D} \ell \raisebox{\depth}{$\chi$} \,dx +  \int_{D} \raisebox{\depth}{$\chi$} \left(\lambda \operatorname{div}\left( {\mathbf{ u }} \right)^{2} + 2 \mu \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \pmb \varepsilon({\mathbf{ u }})^\top  \right)\right) \,dx,\\
\pmb \varepsilon({\mathbf{ u }}) &= \frac{ \nabla {\mathbf{ u }}  +   \nabla {\mathbf{ u }} ^\top }{2}.
 \end{align*}
Let
\begin{equation*}
V_{{\mathbf{ u }}} = \{ {\mathbf{ u }} \in \mathbb P_{2}(D)| {\mathbf{ u }} = \mathtt{\text{0.}}  \quad \text{{on}} \quad\Gamma_{2}\cup\Gamma_{4} \},\end{equation*}
\begin{equation*}
V_{{\mathbf{ v }}} = \{ {\mathbf{ v }} \in \mathbb P_{2}(D)| {\mathbf{ v }} = 0  \quad \text{{on}} \quad\Gamma_{2}\cup\Gamma_{4} \}.\end{equation*}
The state $\left[ {\mathbf{ u }}\right] \in  V_{{\mathbf{ u }}}$ is the solution of the linear problem
\begin{dmath*}
-  \int_{\Gamma_N} f \left( \mathbf e_y \cdot {\mathbf{ v }} \right) \,ds_x +  \int_{D} \raisebox{\depth}{$\chi$} \left(\lambda \operatorname{div} {\mathbf{ v }}  \,\operatorname{tr}\left(\pmb \varepsilon({\mathbf{ u }})\right)  + 2 \mu \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \nabla {\mathbf{ v }}  \right)\right) \,dx = 0\qquad \forall\; {{\mathbf{ v }}\in V_{{\mathbf{ v }}}} . 
\end{dmath*}
The adjoint state $\left[ \tilde{\mathbf{ u }}\right] \in  V_{{\mathbf{ v }}}$ to the direct state $\left[ {\mathbf{ u }}\right]$ is the solution of
\begin{dmath*}
\partial_{({\mathbf{ u }},{\mathbf{ v }})} J({\mathbf{ u }},{\mathbf{ v }}) +  \int_{D} \raisebox{\depth}{$\chi$} \left(\lambda \operatorname{div} \tilde{\mathbf{ u }}  \,\operatorname{tr}\left(\partial_{({\mathbf{ u }},{\mathbf{ v }})} \pmb \varepsilon({\mathbf{ v }})\right)  + 2 \mu \left( \partial_{({\mathbf{ u }},{\mathbf{ v }})} \pmb \varepsilon({\mathbf{ v }}) \mathrel{:}  \nabla \tilde{\mathbf{ u }}  \right)\right) \,dx = 0\qquad \forall\; {{\mathbf{ v }}\in V_{{\mathbf{ u }}}} , 
\end{dmath*}
with 
\begin{align*}
\partial_{({\mathbf{ u }},{\mathbf{ v }})} J({\mathbf{ u }},{\mathbf{ v }}) &=  \int_{D} 2 \raisebox{\depth}{$\chi$} \left(\lambda \operatorname{div}\left( {\mathbf{ u }} \right)^{2} \operatorname{div} {\mathbf{ v }} \operatorname{div}\left( {\mathbf{ u }} \right)^{-1} + \mu \left(\left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \pmb \varepsilon({\mathbf{ v }})^\top  \right) + \left( \pmb \varepsilon({\mathbf{ v }}) \mathrel{:}  \pmb \varepsilon({\mathbf{ u }})^\top  \right)\right)\right) \,dx.
 \end{align*}
In order to compute the shape derivative, we consider a shape pertubation $\mathbf x = T_t(\mathbf X) = \mathbf X + t \mathbf  V(X)$ with a suitable velocity field $\mathbf V$. The perturbed Lagrangian $\mathcal G$ using the Lagrangian state $\left[ {\mathbf{ u }}\right]^{L}$ is given by
\begin{dmath*}
\mathcal G(t,\left[ {\mathbf{ u }}\right]^{L},\left[ {\mathbf{ v }}\right]^{L}) = \mathcal V(J) -  \int_{\Gamma_N} f  \operatorname{det}\mathbf F( t )  \sqrt{\left( \left( \mathbf n \cdot  \mathbf F^{-1}(t) \right) \cdot \left( \mathbf n \cdot  \mathbf F^{-1}(t) \right) \right)} \left( \mathbf e_y \cdot {\mathbf{ v }} \right) \,ds_x +  \int_{D} \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \left(\lambda  \,\operatorname{tr}\left(\left(  \nabla {\mathbf{ v }}  \cdot  \mathbf F^{-1}(t) \right)\right)   \,\operatorname{tr}\left(\mathcal V(\pmb \varepsilon)\right)  + 2 \mu \left( \mathcal V(\pmb \varepsilon) \mathrel{:} \left( \left(  \nabla {\mathbf{ v }}  \cdot  \mathbf F^{-1}(t) \right) \right)\right)\right) \,dx , 
\end{dmath*}
with
\begin{dmath*}
\mathcal V(\pmb \varepsilon) = \frac{\left(  \nabla {\mathbf{ u }}  \cdot  \mathbf F^{-1}(t) \right) +  \left( \left(  \nabla {\mathbf{ u }}  \cdot  \mathbf F^{-1}(t) \right) \right)^\top }{2},
 \end{dmath*}
\begin{dmath*}
\mathcal V(J) =  \int_{D} \ell \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \,dx +  \int_{D} \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \left(\lambda  \,\operatorname{tr}\left(\left(  \nabla {\mathbf{ u }}  \cdot  \mathbf F^{-1}(t) \right)\right)^{2}  + 2 \mu \left( \mathcal V(\pmb \varepsilon) \mathrel{:}  \mathcal V(\pmb \varepsilon)^\top  \right)\right) \,dx.
 \end{dmath*}
For the direct state $\left[ {\mathbf{ u }}\right]$ and the adjoint state $\left[ \tilde{\mathbf{ u }}\right]$, the volume expression of the shape derivative is given by
\begin{dmath*}
DJ(\Omega)(\mathbf V) = - ( \int_{\Gamma_N} f \operatorname{div} {\mathbf V} \left( \mathbf e_y \cdot \tilde{\mathbf{ u }} \right) \,ds_x -  \int_{\Gamma_N} f \left( \mathbf n \cdot \left( \mathbf n \cdot  \nabla {\mathbf V}  \right) \right) \left( \mathbf e_y \cdot \tilde{\mathbf{ u }} \right) \,ds_x) + \partial_{t} J({\mathbf V},{\mathbf{ u }}) +  \int_{D} \raisebox{\depth}{$\chi$} \left(\lambda \operatorname{div} \tilde{\mathbf{ u }}  \,\operatorname{tr}\left(\partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }})\right)  + \lambda  \,\operatorname{tr}\left(- \left(  \nabla \tilde{\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right)\right)   \,\operatorname{tr}\left(\pmb \varepsilon({\mathbf{ u }})\right)  + 2 \mu \left(\left( \partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }}) \mathrel{:}  \nabla \tilde{\mathbf{ u }}  \right) - \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:} \left( \left(  \nabla \tilde{\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right) \right)\right)\right) + \operatorname{div} {\mathbf V} \left(\lambda \operatorname{div} \tilde{\mathbf{ u }}  \,\operatorname{tr}\left(\pmb \varepsilon({\mathbf{ u }})\right)  + 2 \mu \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \nabla \tilde{\mathbf{ u }}  \right)\right)\right) \,dx , 
\end{dmath*}
with
\begin{dmath*}
\partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }}) = - \frac{\left(  \nabla {\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right) +  \left( \left(  \nabla {\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right) \right)^\top }{2},
 \end{dmath*}
\begin{dmath*}
\partial_{t} J({\mathbf V},{\mathbf{ u }}) =  \int_{D} \ell \raisebox{\depth}{$\chi$} \operatorname{div} {\mathbf V} \,dx +  \int_{D} \raisebox{\depth}{$\chi$} \left(\operatorname{div} {\mathbf V} \left(\lambda \operatorname{div}\left( {\mathbf{ u }} \right)^{2} + 2 \mu \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \pmb \varepsilon({\mathbf{ u }})^\top  \right)\right) + 2 \left(\lambda \operatorname{div}\left( {\mathbf{ u }} \right)^{2}  \,\operatorname{tr}\left(- \left(  \nabla {\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right)\right)  \operatorname{div}\left( {\mathbf{ u }} \right)^{-1} + \mu \left(\left( \partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }}) \mathrel{:}  \pmb \varepsilon({\mathbf{ u }})^\top  \right) + \left( \pmb \varepsilon({\mathbf{ u }}) \mathrel{:}  \partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }})^\top  \right)\right)\right)\right) \,dx.
 \end{dmath*}
