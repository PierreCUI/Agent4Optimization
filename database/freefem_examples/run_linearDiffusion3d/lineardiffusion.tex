Let $\phi(x)$ be the level-set function and
 \begin{align*}\raisebox{\depth}{$\chi$}(x) &= \begin{cases} 1 & \mbox{if}\quad \phi(x) < 0 \\ 1/10 & \mbox{if}\quad \phi(x)  \ge 0 \end{cases}.\end{align*}
The Lagrangian of the linear diffusion problem is
\begin{dmath*}
\mathcal L(\left[ u\right],\left[ \delta u\right]) = J(u) -  \int_{D} f \delta u \,dx +  \int_{D} \lambda \raisebox{\depth}{$\chi$} \left(  \nabla u  \cdot  \nabla \delta u  \right) \,dx , 
\end{dmath*}
with 
\begin{align*}
J(u) &=  \int_{D} \ell \raisebox{\depth}{$\chi$} \,dx +  \int_{D} \lambda \raisebox{\depth}{$\chi$} \left(  \nabla u  \cdot  \nabla u  \right) \,dx.
 \end{align*}
Let
\begin{equation*}
V_{u} = \{ u \in \mathbb P_{2}(D)| u = 0  \quad \text{{on}} \quad\Gamma_{3} \},\end{equation*}
\begin{equation*}
V_{\delta u} = \{ \delta u \in \mathbb P_{2}(D)| \delta u = 0  \quad \text{{on}} \quad\Gamma_{3} \}.\end{equation*}
The state $\left[ u\right] \in  V_{u}$ is the solution of the linear problem
\begin{dmath*}
-  \int_{D} f \delta u \,dx +  \int_{D} \lambda \raisebox{\depth}{$\chi$} \left(  \nabla u  \cdot  \nabla \delta u  \right) \,dx = 0\qquad \forall\; {\delta u\in V_{\delta u}} . 
\end{dmath*}
The adjoint state $\left[ \tilde u\right] \in  V_{\delta u}$ to the direct state $\left[ u\right]$ is the solution of
\begin{dmath*}
\partial_{(u,\delta u)} J(\delta u,u) +  \int_{D} \lambda \raisebox{\depth}{$\chi$} \left(  \nabla \delta u  \cdot  \nabla \tilde u  \right) \,dx = 0\qquad \forall\; {\delta u\in V_{u}} , 
\end{dmath*}
with 
\begin{align*}
\partial_{(u,\delta u)} J(\delta u,u) &=  \int_{D} 2 \lambda \raisebox{\depth}{$\chi$} \left(  \nabla u  \cdot  \nabla \delta u  \right) \,dx.
 \end{align*}
In order to compute the shape derivative, we consider a shape pertubation $\mathbf x = T_t(\mathbf X) = \mathbf X + t \mathbf  V(X)$ with a suitable velocity field $\mathbf V$. The perturbed Lagrangian $\mathcal G$ using the Lagrangian state $\left[ u\right]^{L}$ is given by
\begin{dmath*}
\mathcal G(t,\left[ u\right]^{L},\left[ \delta u\right]^{L}) = \mathcal V(J) -  \int_{D} f  \operatorname{det}\mathbf F( t )  \delta u \,dx +  \int_{D} \lambda \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \left( \left(  \nabla u  \cdot  \mathbf F^{-1}(t) \right) \cdot \left(  \nabla \delta u  \cdot  \mathbf F^{-1}(t) \right) \right) \,dx , 
\end{dmath*}
with
\begin{dmath*}
\mathcal V(J) =  \int_{D} \ell \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \,dx +  \int_{D} \lambda \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \left( \left(  \nabla u  \cdot  \mathbf F^{-1}(t) \right) \cdot \left(  \nabla u  \cdot  \mathbf F^{-1}(t) \right) \right) \,dx.
 \end{dmath*}
For the direct state $\left[ u\right]$ and the adjoint state $\left[ \tilde u\right]$, the volume expression of the shape derivative is given by
\begin{dmath*}
DJ(\Omega)(\mathbf V) = \partial_{t} J(u,{\mathbf V}) +  \int_{D} \left(\operatorname{div} {\mathbf V} \left(  \nabla u  \cdot  \nabla \tilde u  \right) - \left(  \nabla \tilde u  \cdot \left(  \nabla u  \cdot  \nabla {\mathbf V}  \right) \right) - \left(  \nabla u  \cdot \left(  \nabla \tilde u  \cdot  \nabla {\mathbf V}  \right) \right)\right) \lambda \raisebox{\depth}{$\chi$} \,dx -  \int_{D} f \operatorname{div} {\mathbf V} \tilde u \,dx , 
\end{dmath*}
with
\begin{dmath*}
\partial_{t} J(u,{\mathbf V}) =  \int_{D} \left(\operatorname{div} {\mathbf V} \left(  \nabla u  \cdot  \nabla u  \right) - 2 \left(  \nabla u  \cdot \left(  \nabla u  \cdot  \nabla {\mathbf V}  \right) \right)\right) \lambda \raisebox{\depth}{$\chi$} \,dx +  \int_{D} \ell \raisebox{\depth}{$\chi$} \operatorname{div} {\mathbf V} \,dx.
 \end{dmath*}
