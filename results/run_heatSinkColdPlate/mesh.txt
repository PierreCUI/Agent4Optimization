real L = 1.0;    // normalized design-domain width
real H = 1.0;    // normalized design-domain height
real lq = 0.2;   // half-length of the load segment (load = central 40% of the bottom edge)
border C01(t=0, L/2-lq){x=t; y=0; label=1;}         // bottom edge, adiabatic left
border C02(t=0, 2*lq){x=L/2-lq+t; y=0; label=2;}    // bottom edge, load segment
border C03(t=0, L/2-lq){x=L/2+lq+t; y=0; label=1;}  // bottom edge, adiabatic right
border C04(t=0, H){x=L; y=t; label=3;}              // right edge, air convection
border C05(t=0, L){x=L-t; y=H; label=4;}            // top edge, cooling-pipe wall (isothermal)
border C06(t=0, H){x=0; y=H-t; label=3;}            // left edge, air convection
int n = 50;
mesh Th = buildmesh(C01(n) + C02(n) + C03(n) + C04(n) + C05(n) + C06(n));
