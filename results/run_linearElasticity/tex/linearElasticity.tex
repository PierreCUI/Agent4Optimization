Let $\phi(x)$ be the level-set function and
 \begin{align*}\raisebox{\depth}{$\chi$}(x) &= \begin{cases} 1 & \mbox{if}\quad \phi(x) < 0 \\ 1/100 & \mbox{if}\quad \phi(x)  \ge 0 \end{cases}.\end{align*}
The Lagrangian of the linear Elasticity problem is
\begin{dmath*}
\mathcal L(\left[ {\mathbf{ u }}\right],\left[ {\delta\mathbf{  u }}\right]) = J({\mathbf{ u }}) -  \int_{\partial D} f \left( \mathbf e_y \cdot {\delta\mathbf{  u }} \right) \,ds_x +  \int_{D} \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:}  \nabla {\delta\mathbf{  u }}  \right) \,dx , 
\end{dmath*}
with 
\begin{align*}
J({\mathbf{ u }}) &=  \int_{D} \ell \raisebox{\depth}{$\chi$} \,dx +  \int_{D} \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:} \pmb \varepsilon({\mathbf{ u }}) \right) \,dx,\\
\pmb \sigma({\mathbf{ u }}) &= \raisebox{\depth}{$\chi$} \left(\mathbf{I} \lambda  \,\operatorname{tr}\left(\pmb \varepsilon({\mathbf{ u }})\right)  + 2 \mu \pmb \varepsilon({\mathbf{ u }})\right),\\
\pmb \varepsilon({\mathbf{ u }}) &= \frac{ \nabla {\mathbf{ u }}  +   \nabla {\mathbf{ u }} ^\top }{2},\\
\mu &= \frac{E}{2 \left(\nu + 1\right)},\\
\lambda &= - \frac{E \nu}{\left(\nu + 1\right) \left(2 \nu - 1\right)}.
 \end{align*}
Let
\begin{equation*}
V_{{\mathbf{ u }}} = \{ {\mathbf{ u }} \in \mathbb P_{2}(D)| {\mathbf{ u }} = \mathtt{\text{0.}}  \quad \text{{on}} \quad\Gamma_{4} \},\end{equation*}
\begin{equation*}
V_{{\delta\mathbf{  u }}} = \{ {\delta\mathbf{  u }} \in \mathbb P_{2}(D)| {\delta\mathbf{  u }} = 0  \quad \text{{on}} \quad\Gamma_{4} \}.\end{equation*}
The state $\left[ {\mathbf{ u }}\right] \in  V_{{\mathbf{ u }}}$ is the solution of the linear problem
\begin{dmath*}
-  \int_{\partial D} f \left( \mathbf e_y \cdot {\delta\mathbf{  u }} \right) \,ds_x +  \int_{D} \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:}  \nabla {\delta\mathbf{  u }}  \right) \,dx = 0\qquad \forall\; {{\delta\mathbf{  u }}\in V_{{\delta\mathbf{  u }}}} . 
\end{dmath*}
The adjoint state $\left[ \tilde{\mathbf{ u }}\right] \in  V_{{\delta\mathbf{  u }}}$ to the direct state $\left[ {\mathbf{ u }}\right]$ is the solution of
\begin{dmath*}
\partial_{({\mathbf{ u }},{\delta\mathbf{  u }})} J({\delta\mathbf{  u }},{\mathbf{ u }}) +  \int_{D} \left( \partial_{({\mathbf{ u }},{\delta\mathbf{  u }})} \pmb \sigma({\delta\mathbf{  u }}) \mathrel{:}  \nabla \tilde{\mathbf{ u }}  \right) \,dx = 0\qquad \forall\; {{\delta\mathbf{  u }}\in V_{{\mathbf{ u }}}} , 
\end{dmath*}
with 
\begin{align*}
\partial_{({\mathbf{ u }},{\delta\mathbf{  u }})} J({\delta\mathbf{  u }},{\mathbf{ u }}) &=  \int_{D} \left( \pmb \sigma({\delta\mathbf{  u }}) \mathrel{:} \pmb \varepsilon({\mathbf{ u }}) \right) + \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:} \pmb \varepsilon({\delta\mathbf{  u }}) \right) \,dx.
 \end{align*}
In order to compute the shape derivative, we consider a shape pertubation $\mathbf x = T_t(\mathbf X) = \mathbf X + t \mathbf  V(X)$ with a suitable velocity field $\mathbf V$. The perturbed Lagrangian $\mathcal G$ using the Lagrangian state $\left[ {\mathbf{ u }}\right]^{L}$ is given by
\begin{dmath*}
\mathcal G(t,\left[ {\mathbf{ u }}\right]^{L},\left[ {\delta\mathbf{  u }}\right]^{L}) = \mathcal V(J) -  \int_{\partial D} f  \operatorname{det}\mathbf F( t )  \sqrt{\left( \left( \mathbf n \cdot  \mathbf F^{-1}(t) \right) \cdot \left( \mathbf n \cdot  \mathbf F^{-1}(t) \right) \right)} \left( \mathbf e_y \cdot {\delta\mathbf{  u }} \right) \,ds_x +  \int_{D}  \operatorname{det}\mathbf F( t )  \left( \mathcal V(\pmb \sigma) \mathrel{:} \left( \left(  \nabla {\delta\mathbf{  u }}  \cdot  \mathbf F^{-1}(t) \right) \right)\right) \,dx , 
\end{dmath*}
with
\begin{dmath*}
\mathcal V(J) =  \int_{D} \ell \raisebox{\depth}{$\chi$}  \operatorname{det}\mathbf F( t )  \,dx +  \int_{D}  \operatorname{det}\mathbf F( t )  \left( \mathcal V(\pmb \sigma) \mathrel{:} \mathcal V(\pmb \varepsilon) \right) \,dx,
 \end{dmath*}
\begin{dmath*}
\mathcal V(\pmb \sigma) = \raisebox{\depth}{$\chi$} \left(\mathbf{I} \lambda  \,\operatorname{tr}\left(\mathcal V(\pmb \varepsilon)\right)  + 2 \mu \mathcal V(\pmb \varepsilon)\right),
 \end{dmath*}
\begin{dmath*}
\mathcal V(\pmb \varepsilon) = \frac{\left(  \nabla {\mathbf{ u }}  \cdot  \mathbf F^{-1}(t) \right) +  \left( \left(  \nabla {\mathbf{ u }}  \cdot  \mathbf F^{-1}(t) \right) \right)^\top }{2}.
 \end{dmath*}
For the direct state $\left[ {\mathbf{ u }}\right]$ and the adjoint state $\left[ \tilde{\mathbf{ u }}\right]$, the volume expression of the shape derivative is given by
\begin{dmath*}
DJ(\Omega)(\mathbf V) = - ( \int_{\partial D} f \operatorname{div} {\mathbf V} \left( \mathbf e_y \cdot \tilde{\mathbf{ u }} \right) \,ds_x -  \int_{\partial D} f \left( \mathbf n \cdot \left( \mathbf n \cdot  \nabla {\mathbf V}  \right) \right) \left( \mathbf e_y \cdot \tilde{\mathbf{ u }} \right) \,ds_x) + \partial_{t} J({\mathbf V},{\mathbf{ u }}) +  \int_{D} \operatorname{div} {\mathbf V} \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:}  \nabla \tilde{\mathbf{ u }}  \right) + \left( \partial_{t} \pmb \sigma({\mathbf V},{\mathbf{ u }}) \mathrel{:}  \nabla \tilde{\mathbf{ u }}  \right) - \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:} \left( \left(  \nabla \tilde{\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right) \right)\right) \,dx , 
\end{dmath*}
with
\begin{dmath*}
\partial_{t} \pmb \sigma({\mathbf V},{\mathbf{ u }}) = \raisebox{\depth}{$\chi$} \left(\mathbf{I} \lambda  \,\operatorname{tr}\left(\partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }})\right)  + 2 \mu \partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }})\right),
 \end{dmath*}
\begin{dmath*}
\partial_{t} J({\mathbf V},{\mathbf{ u }}) =  \int_{D} \ell \raisebox{\depth}{$\chi$} \operatorname{div} {\mathbf V} \,dx +  \int_{D} \operatorname{div} {\mathbf V} \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:} \pmb \varepsilon({\mathbf{ u }}) \right) + \left( \partial_{t} \pmb \sigma({\mathbf V},{\mathbf{ u }}) \mathrel{:} \pmb \varepsilon({\mathbf{ u }}) \right) + \left( \pmb \sigma({\mathbf{ u }}) \mathrel{:} \partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }}) \right) \,dx,
 \end{dmath*}
\begin{dmath*}
\partial_{t} \pmb \varepsilon({\mathbf V},{\mathbf{ u }}) = - \frac{\left(  \nabla {\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right) +  \left( \left(  \nabla {\mathbf{ u }}  \cdot  \nabla {\mathbf V}  \right) \right)^\top }{2}.
 \end{dmath*}
